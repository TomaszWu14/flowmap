Błąd AI → zapis oceny z reguł + komunikat.

**Tech Stack:** Flask, Anthropic SDK (`call_tool`), Pydantic, Jinja, pytest (podrobiony `_client`), Playwright.

## Global Constraints

- Gałąź `feat/trainer-ai-score`; PR → 5 kontroli CI → merge → wdrożenie → produkcja: `POST /p/1/ocena/ai` z 404 na 400 (CSRF).
- Kryteria AI (kod → etykieta): `naming` „Nazewnictwo”, `business` „Sens biznesowy”, `exceptions` „Wyjątki branżowe”, `decisions` „Jakość decyzji”; każde dokładnie raz, `score` 0–5, `reason` niepusty.

`app/trainer/ai_review.py`:

```python
"""Qualitative AI assessment (Opus): four 0–5 criteria with reasons and one top fix; 30% of the mixed score."""

import json
from typing import Literal

from pydantic import BaseModel, field_validator
from pydantic import Field as PField

from ..ai.client import _obj, _prompt, call_tool, compact_graph

LABELS = {"naming": "Nazewnictwo", "business": "Sens biznesowy", "exceptions": "Wyjątki branżowe",
          "decisions": "Jakość decyzji"}
TOOL = {"name": "assess_quality", "strict": True, "description": "Jakościowa ocena mapy procesu i projektu aplikacji.",
        "input_schema": _obj({
            "criteria": {"type": "array", "items": _obj({"code": {"type": "string", "enum": list(LABELS)},
                                                          "score": {"type": "integer"},
                                                          "reason": {"type": "string"}})},
            "top_fix": {"type": "string"}})}


class _Criterion(BaseModel):
    code: Literal["naming", "business", "exceptions", "decisions"]
    score: int = PField(ge=0, le=5)
    reason: str = PField(min_length=1, max_length=600)


class _Assessment(BaseModel):
    criteria: list[_Criterion]
    top_fix: str = PField(min_length=1, max_length=600)

    @field_validator("criteria")
    @classmethod
    def each_once(cls, v):
        if sorted(c.code for c in v) != sorted(LABELS):
            raise ValueError("każde kryterium dokładnie raz")
        return v


def assess(project, graph: dict, spec, results: list[dict]) -> dict:
    """One Opus call. Raises AIError (from call_tool) when the model fails twice."""
    spec_json = json.dumps(spec.model_dump(mode="json", exclude={"proposals"}), ensure_ascii=False) if spec else "{}"
    summary = "; ".join(f"{r['label']} {r['score']}/100" for r in results)
    result: _Assessment = call_tool(_prompt("assess", start_sentence=project.start_sentence,
                                            graph_json=compact_graph(graph), spec_json=spec_json,
                                            rules_summary=summary), TOOL, _Assessment)
    criteria = [{"code": c.code, "label": LABELS[c.code], "score": c.score, "reason": c.reason}
                for c in sorted(result.criteria, key=lambda c: list(LABELS).index(c.code))]
    return {"criteria": criteria, "top_fix": result.top_fix, "pct": round(100 * sum(c.score for c in result.criteria) / 20)}
```

`app/trainer/rubric.py`:

```python
AI_WEIGHT = 0.3


def mix(results: list[dict], ai: dict) -> list[dict]:
    """Rules stay visible as rules_score; the AI part is 30% of the shown score."""
    return [r | {"rules_score": r["score"], "ai": ai,
                 "score": round((1 - AI_WEIGHT) * r["score"] + AI_WEIGHT * ai["pct"])} for r in results]
```

@bp.post("/p/<int:project_id>/ocena/ai")
@login_required
def score_ai(project_id):
    project = g.project_for(project_id, "view")
    g.ensure_spec(project)
    results = evaluate(project)
    reason = spend_ai_request()
    if reason:
        flash(f"{reason} — zapisano ocenę z reguł.", "error")
        _save(project, results, "rules")
    else:
        try:
            ai = assess(project, project.graph(), g.project_spec(project), results)
            _save(project, mix(results, ai), "mixed")
        except AIError as exc:
            flash(f"Ocena AI nieudana ({exc}) — zapisano ocenę z reguł.", "error")
            _save(project, results, "rules")
    return redirect(url_for("trainer.score", project_id=project.id))
